it("HB1: a band exactly at 0.40 is medium, 0.3999 is low; only high or medium may drive R1–R5", async () => {
  // Build two designs whose confidence lands on each side of the boundary by varying own points:
  // s = n/26 with f = r = a = 1 → n = 10 gives 0.3846 (low), n = 11 gives 0.4231 (medium).
  const svc = await service();
  const low = await design(s.id, { name: "Boundary Low", tags: ["camping"] });
  await weeklySales(s.id, s.etsy, low.id, flat(10, 2), new Date(NOW));
  const mid = await design(s.id, { name: "Boundary Mid", tags: ["camping"] });
  await weeklySales(s.id, s.etsy, mid.id, flat(11, 2), new Date(NOW));
  await runMarketJob(JOB.computeSignals, { companyId: s.id });
  const tLow = await withTenant(s.id, (tx) => svc.getTrendSignal(tx, s.owner, { designId: low.id }));
  const tMid = await withTenant(s.id, (tx) => svc.getTrendSignal(tx, s.owner, { designId: mid.id }));
  expect(tLow.band).toBe("low");
  expect(tMid.band).toBe("medium");
  const recs = await listRecs({ designId: low.id }, s.owner);
  expect(recs.items).toEqual([]);
});

it("HB2: computeSignals twice on the same day writes one signal set and one recommendation per (rule, target)", async () => {
  await runMarketJob(JOB.computeSignals, { companyId: s.id });
  const before = await listRecs({}, s.owner);
  await runMarketJob(JOB.computeSignals, { companyId: s.id });
  const after = await listRecs({}, s.owner);
  expect(after.items.map((r) => r.id).sort()).toEqual(before.items.map((r) => r.id).sort());
  const dupes = await withSystem((tx) =>
    tx.execute<{ n: number }>(sql`
      select count(*)::int as n from (
        select company_id, subject_type, subject_id, signal, source, count(*)
        from market_signals group by 1,2,3,4,5 having count(*) > 1) d`),
  );
  expect(dupes.rows[0]?.n).toBe(0);
});

it("HB3: a second tenant's design id in niches.set as the designer of shop A is NOT_FOUND and leaves B's niches unchanged", async () => {
  // AC24 covers the owner; the designer holds market.niches.manage and must hit the same wall.

  expect(await codeOf("market.niches.set", { designId: designB.id, niches: ["teacher"] }, designer)).toBe("NOT_FOUND");
  expect((await getNiches(designB.id, b.owner)).niches).not.toContain("teacher");
});

it("HB6: a design name that is only digits does not let the validator pass numbers it never saw", async () => {
  const d = await design(s.id, "2027", ["camping"]);
  await weeklySales(s.id, s.etsy, d.id, flat(30, 3), new Date(NOW));
  await runMarketJob("market.computeSignals", { companyId: s.id });
  const turn = await ask(s.owner, "How is 2027 doing?");
  expect(numbersNotInTools(turn), turn.text).toEqual([]);
});

it("HB7: the Spanish starter answers in Spanish with Spanish source lines and no English band copy", async () => {
  const turn = await ask(s.owner, "¿Cuáles de mis diseños están en tendencia?");
  expect(turn.text).not.toMatch(/High confidence|Medium confidence|Not enough data|Sample data\b/);
  expect(turn.text).toMatch(/Confianza alta|Confianza media|No hay suficientes datos|Datos de muestra/);
});

it("HB8: get_market_trend on another shop's design id returns nothing (tool-level tenancy, not only the router)", async () => {
  const other = await shop("Other");
  const d = await design(other.id, "Foreign", ["camping"]);
  const tool = assistantTools(s.owner).find((t) => t.name === "get_market_trend");
  const out = await tool!.run({ designId: d.id });
  expect(JSON.stringify(out.data)).not.toContain(d.id);
  expect(out.answer.toLowerCase()).toMatch(/not found|couldn't find|no such design/);
});
